pip

镜像源 

pip config set global.index-url https://mirrors.aliyun.com/pypi/simple

寻找pip 

pip show pip

安装深度学习环境

1. numpy先行 

pip install numpy==1.26.4

2. numpy的附属包 

pip install scipy==1.13.1
pip install pandas==2.2.2
pip install matplotlib==3.9.2
pip install seaborn==0.13.2
pip install scikit-learn==1.5.1

3. pytorch 

pip install torch==2.7.0+cu118 torchvision==0.22.0+cu118 torchaudio==2.7.0+cu118 \
  --index-url https://download.pytorch.org/whl/cu118

一键sh 

#!/bin/bash

echo "【第1步】检查 pip 安装位置"
pip show pip

read -p "是否继续安装环境？(y/n): " confirm
if [ "$confirm" != "y" ]; then
    echo "已取消"
    exit 0
fi

echo "【第2步】检查默认镜像源"
# default_pypi="https://pypi.org/simple/"
# default_pypi="https://pypi.tuna.tsinghua.edu.cn/simple/"
default_pypi="https://mirrors.aliyun.com/pypi/simple"
read -p "请输入pip镜像源（默认：$default_pypi）：" pypi_source
# 如果用户直接回车，使用默认源default_pypi
if [ -z "$pypi_source" ]; then
    pypi_source="$default_pypi"
fi
echo "使用pypi镜像源：$pypi_source"

echo "【第3步】检查torch镜像源"
# default_torch="https://download.pytorch.org/whl/cu118"
default_torch="https://mirror.sjtu.edu.cn/pytorch-wheels/cu118"
read -p "请输入torch镜像源（默认：$default_torch）：" torch_source
# 如果用户直接回车，使用默认源default_torch
if [ -z "$torch_source" ]; then
    torch_source="$default_torch"
fi
echo "使用torch镜像源：$torch_source"

echo "【第4步】开始安装 - basic"
pip install pip -i "$pypi_source"
pip install twine -i "$pypi_source"
pip install tqdm -i "$pypi_source"
pip install pyyaml -i "$pypi_source"
pip install python-dotenv -i "$pypi_source"

echo "【第4步】开始安装 - numpy"
pip install numpy==1.26.4 -i "$pypi_source"

echo "【第4步】开始安装 - numpy 的附属"
pip install scipy==1.13.1 -i "$pypi_source"
pip install pandas==2.2.2 -i "$pypi_source"
pip install matplotlib==3.9.2 -i "$pypi_source"
pip install seaborn==0.13.2 -i "$pypi_source"
pip install scikit-learn==1.5.1 -i "$pypi_source"

echo "【第4步】开始安装 - torch"
pip install torch==2.7.0+cu118 -i "$torch_source"
pip install torchvision==0.22.0+cu118 -i "$torch_source"
pip install torchaudio==2.7.0+cu118 -i "$torch_source"

echo "【第4步】开始安装 - spikingjelly"
pip install spikingjelly==0.0.0.0.14 -i "$pypi_source"
pip install tensorboard -i "$pypi_source"
pip install cupy-cuda12x -i "$pypi_source"

echo "【第4步】开始安装 - cv"
pip install opencv-python==4.10.0.84 -i "$pypi_source"

echo "【第4步】开始安装 - byzh"
pip install byzh-core -i "$pypi_source"
pip install byzh-ai -i "$pypi_source"
pip install byzh-extra -i "$pypi_source"

echo "【第4步】开始安装 - else"
pip install gradio -i "$pypi_source"
