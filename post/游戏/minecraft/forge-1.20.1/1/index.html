第一个物品[Items, Item]

源码

带s的一般是用来注册的

  • 双shift->items(类那栏)->net.minecraft.world.items->items.java
  • ctrl+b进入item.java
  • ctrl+h查看item的各种子类

我们forge用的是自己的延迟注册系统, 不用管原版是怎么注册的

ResourceLocation.java

ResourceLocation.java 提到了, 各种注册名name只能用:

  • 小写字母
  • 0-9
  • .
  • _
  • /

否则会抛出异常 ResourceLocationException

ResourceLocation.java 还提到了, 如果命名空间不写的话, 默认就是minecraft:xxx

com/byzh/diverse_draw/item/Moditems.java 

package com.byzh.diverse_draw.item;

import com.byzh.diverse_draw.DiverseDraw;
import net.minecraft.world.item.Item;
import net.minecraft.world.level.block.Block;
import net.minecraftforge.eventbus.api.IEventBus;
import net.minecraftforge.registries.DeferredRegister;
import net.minecraftforge.registries.ForgeRegistries;
import net.minecraftforge.registries.RegistryObject;

import java.util.function.Supplier;

public class ModItems {
    // 延迟注册器
    public static final DeferredRegister<Item> ITEMS =
            DeferredRegister.create(ForgeRegistries.ITEMS, DiverseDraw.MODID);

    // 使用`延迟注册器`注册一个对象
    public static final RegistryObject<Item> ICE_ETHER =
            registerItem("ice_ether", () -> new Item(new Item.Properties()));
    public static final RegistryObject<Item> RAW_ICE_ETHER =
            registerItem("raw_ice_ether", () -> new Item(new Item.Properties()));
    public static final RegistryObject<Item> CARDBOARD = // 斜杆是用来区分文件夹的, 最后会被转译成`.`
            registerItem("material/cardboard", () -> new Item(new Item.Properties()));

    // `注册物品`
    private static <T extends Item> RegistryObject<T> registerItem(String name, Supplier<T> item) {
        return ITEMS.register(name, item);
    }

    public static void register(IEventBus eventBus) {
        ITEMS.register(eventBus);
    }
}

在构造方法里写上: Moditems.register(modEventBus);

如何把物品加入原版的物品栏里

修改主文件 DiverseDraw.java 里的 addCreative 方法:

private void addCreative(BuildCreativeModeTabContentsEvent event)
    {
        if (event.getTabKey() == CreativeModeTabs.BUILDING_BLOCKS) {
            event.accept(Moditems.ICE_ETHER);
            event.accept(Moditems.RAW_ICE_ETHER);
            event.accept(Moditems.CARDBOARD);
        }
//        if (event.getTabKey() == CreativeModeTabs.BUILDING_BLOCKS)
//            event.accept(EXAMPLE_BLOCK_ITEM);
    }
